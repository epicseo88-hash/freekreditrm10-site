function formatRM(value) {
  return "RM" + Number(value).toLocaleString("en-MY", {
    minimumFractionDigits: 2,
    maximumFractionDigits: 2
  });
}

function calculateWagering(event) {
  if (event) {
    event.preventDefault();
  }

  var base = parseFloat(document.getElementById("wageringBase").value);
  var multiplier = parseFloat(document.getElementById("multiplier").value);
  var contribution = parseFloat(document.getElementById("contribution").value);

  if (
    !Number.isFinite(base) ||
    !Number.isFinite(multiplier) ||
    !Number.isFinite(contribution) ||
    base < 0 ||
    multiplier < 0 ||
    contribution <= 0
  ) {
    document.getElementById("calculationExplanation").textContent =
      "Please enter valid non-negative values and select a game contribution above 0%.";
    return false;
  }

  var nominalTurnover = base * multiplier;
  var estimatedWagering = nominalTurnover / contribution;
  var contributionPercentage = contribution * 100;

  document.getElementById("nominalResult").textContent =
    formatRM(nominalTurnover);

  document.getElementById("contributionResult").textContent =
    formatRM(estimatedWagering);

  document.getElementById("calculationExplanation").textContent =
    formatRM(base) +
    " × " +
    multiplier +
    "× = " +
    formatRM(nominalTurnover) +
    " nominal turnover. At " +
    contributionPercentage +
    "% game contribution, approximately " +
    formatRM(estimatedWagering) +
    " of qualifying wagering would be needed to produce " +
    formatRM(nominalTurnover) +
    " of qualifying turnover in this simplified calculation.";

  return false;
}

function applyAmountPreset() {
  var preset = document.getElementById("amountPreset").value;

  if (preset !== "") {
    document.getElementById("wageringBase").value = preset;
    calculateWagering();
  }
}

function resetWageringCalculator() {
  document.getElementById("wageringBase").value = "10";
  document.getElementById("multiplier").value = "10";
  document.getElementById("contribution").value = "1";
  document.getElementById("amountPreset").value = "";
  calculateWagering();
}

document.addEventListener("DOMContentLoaded", function () {
  calculateWagering();
});
